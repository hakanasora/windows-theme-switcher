cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" TS_VERSION LIMIT_COUNT 1)
project(ThemeSwitcher VERSION ${TS_VERSION} LANGUAGES CXX RC)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/VERSION")
# Build-Native.ps1 pins VSLANG so Ninja can reliably identify header dependencies.
set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:")
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Theme Switcher requires native Windows x64.")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "Build with MSVC and the Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
set(TS_VERSION "${PROJECT_VERSION}")
set(TS_ARTIFACTS "${CMAKE_BINARY_DIR}/artifacts")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${TS_ARTIFACTS}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" CONFIG_UPPER)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${TS_ARTIFACTS}")
endforeach()
configure_file(src/native/Version.h.in generated/Version.h @ONLY)
configure_file(src/native/App.manifest.in generated/App.manifest @ONLY)
configure_file(src/native/App.rc.in generated/App.rc @ONLY)
add_library(ThemeCore STATIC src/native/ThemeCore.cpp src/native/ThemeCache.cpp)
target_include_directories(ThemeCore PUBLIC src/native "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(ThemeCore PUBLIC advapi32 user32)
add_executable(ThemeSwitcher WIN32 src/native/Main.cpp "${CMAKE_BINARY_DIR}/generated/App.rc")
target_link_libraries(ThemeSwitcher PRIVATE ThemeCore shell32)
target_link_options(ThemeSwitcher PRIVATE /MANIFEST:NO)
add_library(ThemeSwitcherShell SHARED src/native/ShellExtension.cpp src/native/ShellExtension.def)
target_link_libraries(ThemeSwitcherShell PRIVATE ThemeCore shell32 shlwapi ole32)
set_target_properties(ThemeSwitcherShell PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${TS_ARTIFACTS}/components/${TS_VERSION}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(ThemeSwitcherShell PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${TS_ARTIFACTS}/components/${TS_VERSION}")
endforeach()
add_executable(ThemeSwitcherDeploy WIN32 src/native/Deploy.cpp "${CMAKE_BINARY_DIR}/generated/App.rc")
set_target_properties(ThemeSwitcherDeploy PROPERTIES OUTPUT_NAME "ThemeSwitcher.Deploy")
target_link_libraries(ThemeSwitcherDeploy PRIVATE ThemeCore windowsapp shell32 shlwapi ole32 wintrust crypt32)
target_link_options(ThemeSwitcherDeploy PRIVATE /MANIFEST:NO)
enable_testing()
add_executable(ThemeSwitcherTests tests/NativeTests.cpp)
target_link_libraries(ThemeSwitcherTests PRIVATE ThemeCore shell32 ole32)
add_dependencies(ThemeSwitcherTests ThemeSwitcherShell ThemeSwitcher)
add_test(NAME NativeCoreCacheShell COMMAND ThemeSwitcherTests "$<TARGET_FILE:ThemeSwitcherShell>" "$<TARGET_FILE:ThemeSwitcher>")
set_tests_properties(NativeCoreCacheShell PROPERTIES TIMEOUT 90)
add_executable(ThemeSwitcherDeployTests tests/DeployTests.cpp)
target_link_libraries(ThemeSwitcherDeployTests PRIVATE ThemeCore windowsapp shell32 shlwapi ole32 wintrust crypt32)
add_test(NAME DeploymentIsolation COMMAND ThemeSwitcherDeployTests)
set_tests_properties(DeploymentIsolation PROPERTIES TIMEOUT 30)
